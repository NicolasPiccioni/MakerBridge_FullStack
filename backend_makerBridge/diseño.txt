BASE DE DATOS | ESTRUCTURA SQL

-- ============================================================
-- MakerBridge — Esquema de Base de Datos Relacional (v2)
-- Motor: PostgreSQL
-- Tablas: usuarios, pedidos, cotizaciones, login_historial, pagos
-- ============================================================

-- Limpieza (útil para re-ejecutar el script en desarrollo)
DROP TABLE IF EXISTS pagos CASCADE;
DROP TABLE IF EXISTS cotizaciones CASCADE;
DROP TABLE IF EXISTS login_historial CASCADE;
DROP TABLE IF EXISTS pedidos CASCADE;
DROP TABLE IF EXISTS usuarios CASCADE;

-- ============================================================
-- 1. USUARIOS (clientes, makers y, opcionalmente, admins)
-- ============================================================

CREATE TABLE usuarios (
    id_usuario      SERIAL PRIMARY KEY,
    nombre          VARCHAR(150) NOT NULL,
    email           VARCHAR(150) NOT NULL UNIQUE,
    password_hash   VARCHAR(255) NOT NULL,          -- nunca guardar la contraseña en texto plano
    rol             VARCHAR(10) NOT NULL CHECK (rol IN ('cliente', 'maker', 'admin')),
    activo          BOOLEAN NOT NULL DEFAULT TRUE,
    fecha_registro  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 2. PEDIDOS (creados por un usuario con rol 'cliente')
-- ============================================================

CREATE TABLE pedidos (
    id_pedido       SERIAL PRIMARY KEY,
    id_usuario      INTEGER NOT NULL REFERENCES usuarios(id_usuario) ON DELETE CASCADE,
    descripcion     TEXT NOT NULL,
    cantidad        INTEGER NOT NULL DEFAULT 1,
    material        VARCHAR(50),
    color           VARCHAR(50),
    archivo_url     VARCHAR(300),                    -- STL o imagen de referencia
    fecha_limite    DATE,
    estado          VARCHAR(20) NOT NULL DEFAULT 'pendiente'
                    CHECK (estado IN ('pendiente', 'cotizado', 'en_proceso', 'terminado', 'entregado', 'cancelado')),
    fecha_creacion  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 3. COTIZACIONES (un maker cotiza un pedido)
-- ============================================================

CREATE TABLE cotizaciones (
    id_cotizacion   SERIAL PRIMARY KEY,
    id_pedido       INTEGER NOT NULL REFERENCES pedidos(id_pedido) ON DELETE CASCADE,
    id_maker        INTEGER NOT NULL REFERENCES usuarios(id_usuario),
    precio          DECIMAL(10,2) NOT NULL,
    tiempo_estimado_dias INTEGER,
    mensaje         TEXT,
    estado          VARCHAR(12) NOT NULL DEFAULT 'pendiente'
                    CHECK (estado IN ('pendiente', 'aceptada', 'rechazada')),
    fecha_creacion  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (id_pedido, id_maker)   -- un maker cotiza una sola vez por pedido
);

-- ============================================================
-- 4. LOGIN_HISTORIAL (auditoría de intentos de inicio de sesión)
-- ============================================================

CREATE TABLE login_historial (
    id_historial    SERIAL PRIMARY KEY,
    id_usuario      INTEGER NOT NULL REFERENCES usuarios(id_usuario) ON DELETE CASCADE,
    fecha           TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ip              VARCHAR(45),                     -- soporta IPv4 e IPv6
    exito           BOOLEAN NOT NULL
);

-- ============================================================
-- 5. PAGOS (relacionado con una cotización aceptada)
-- ============================================================

CREATE TABLE pagos (
    id_pago         SERIAL PRIMARY KEY,
    id_cotizacion   INTEGER NOT NULL UNIQUE REFERENCES cotizaciones(id_cotizacion) ON DELETE CASCADE,
    monto           DECIMAL(10,2) NOT NULL,
    metodo_pago     VARCHAR(30) CHECK (metodo_pago IN ('tarjeta', 'transferencia', 'efectivo', 'mercado_pago')),
    estado          VARCHAR(12) NOT NULL DEFAULT 'pendiente'
                    CHECK (estado IN ('pendiente', 'aprobado', 'rechazado', 'reembolsado')),
    fecha_pago      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- ÍNDICES recomendados
-- ============================================================

CREATE INDEX idx_pedidos_usuario       ON pedidos(id_usuario);
CREATE INDEX idx_pedidos_estado        ON pedidos(estado);
CREATE INDEX idx_cotizaciones_pedido   ON cotizaciones(id_pedido);
CREATE INDEX idx_cotizaciones_maker    ON cotizaciones(id_maker);
CREATE INDEX idx_login_historial_user  ON login_historial(id_usuario);
CREATE INDEX idx_pagos_cotizacion      ON pagos(id_cotizacion);
